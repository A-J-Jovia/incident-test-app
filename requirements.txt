# main.py
import uvicorn
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class Item(BaseModel):
    name: str
    price: float
    is_available: bool

@app.get("/items/")
def read_items():
    return [{"name": "Item1", "price": 100.0, "is_available": True}, {"name": "Item2", "price": 200.0, "is_available": False}]

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)
